func evidenceClaim(id, facet string, status model.Status, deps ...string) model.Claim {
	ro := model.RestsNone("leaf")
	if len(deps) > 0 {
		ro = model.RestsOnIDs(deps...)
	}
	return model.Claim{ID: id, Module: "m", Facet: facet, Status: status, Layout: model.LayoutCard, Summary: "s", RestsOn: ro}
}

func evidenceID(i int, internals bool) (string, string) {
	if internals {
		return fmt.Sprintf("m.internals.n%04d", i), "internals"
	}
	return fmt.Sprintf("m.contract.n%04d", i), "contract"
}

// chain: n0 -> n1 -> ... ; odd nodes internals.
func evidenceChain(n int) []model.Claim {
	out := make([]model.Claim, n)
	for i := 0; i < n; i++ {
		id, facet := evidenceID(i, i%2 == 1)
		var deps []string
		if i+1 < n {
			dep, _ := evidenceID(i+1, (i+1)%2 == 1)
			deps = []string{dep}
		}
		out[i] = evidenceClaim(id, facet, model.StatusDraft, deps...)
	}
	return out
}

// dense: L layers x W, each node rests on every node of the next layer;
// odd layers internals.
func evidenceDense(layers, width int) []model.Claim {
	var out []model.Claim
	idx := func(l, w int) int { return l*width + w }
	for l := 0; l < layers; l++ {
		for w := 0; w < width; w++ {
			id, facet := evidenceID(idx(l, w), l%2 == 1)
			var deps []string
			if l+1 < layers {
				for x := 0; x < width; x++ {
					dep, _ := evidenceID(idx(l+1, x), (l+1)%2 == 1)
					deps = append(deps, dep)
				}
			}
			out = append(out, evidenceClaim(id, facet, model.StatusDraft, deps...))
		}
	}
	return out
}

func TestZZGraphSafetyEvidence(t *testing.T) {
	out := os.Getenv("DX_EVIDENCE_OUT")
	if out == "" {
		t.Skip("evidence harness")
	}
	type row struct {
		Shape         string `json:"shape"`
		Claims        int    `json:"claims"`
		Exported      int    `json:"exported"`
		Bytes         int    `json:"catalog_bytes"`
		LeakedIDs     int    `json:"internals_id_occurrences"`
		Redacted      int    `json:"redacted_tokens"`
		OmittedEdges  int    `json:"rests_on_internals_omitted"`
		RecordsIn     int    `json:"records_in"`
		RecordsOut    int    `json:"records_out"`
		BoolsMatch    bool   `json:"readiness_booleans_match"`
		ElapsedMicros int64  `json:"encode_elapsed_us"`
		AllocBytes    uint64 `json:"encode_total_alloc_bytes"`
	}
	var rows []row
	shapes := map[string][]model.Claim{}
	order := []string{}
	for _, n := range []int{32, 64, 128} {
		k := fmt.Sprintf("chain-%d", n)
		shapes[k] = evidenceChain(n)
		order = append(order, k)
	}
	for _, lw := range [][2]int{{8, 5}, {16, 5}, {24, 5}} {
		k := fmt.Sprintf("dense-%dx%d", lw[0], lw[1])
		shapes[k] = evidenceDense(lw[0], lw[1])
		order = append(order, k)
	}
	for _, name := range order {
		claims := shapes[name]
		cat, err := Build(claims, nil)
		if err != nil {
			t.Fatal(err)
		}
		assessments := readiness.Compute(claims, nil, nil)
		cat.SetReadiness(assessments)
		runtime.GC()
		var before, after runtime.MemStats
		runtime.ReadMemStats(&before)
		start := time.Now()
		data, err := EncodeJSONBounded(cat, conformance.MaxOutputBytes)
		elapsed := time.Since(start)
		runtime.ReadMemStats(&after)
		if err != nil {
			t.Fatal(err)
		}
		var doc struct {
			Claims []struct {
				ID    string `json:"id"`
				Edges struct {
					Omitted int `json:"rests_on_internals_omitted"`
				} `json:"edges"`
				Readiness readiness.Assessment `json:"readiness"`
			} `json:"claims"`
		}
		if err := json.Unmarshal(data, &doc); err != nil {
			t.Fatal(err)
		}
		r := row{Shape: name, Claims: len(claims), Exported: len(doc.Claims), Bytes: len(data),
			LeakedIDs: strings.Count(string(data), ".internals."), Redacted: strings.Count(string(data), `"(internals)"`),
			BoolsMatch: true, ElapsedMicros: elapsed.Microseconds(), AllocBytes: after.TotalAlloc - before.TotalAlloc}
		for _, e := range doc.Claims {
			in := assessments[e.ID]
			o := e.Readiness
			r.OmittedEdges += e.Edges.Omitted
			r.RecordsIn += len(in.Conditions) + len(in.DependencyConditions) + len(in.Causes) + len(in.ReviewCauses)
			r.RecordsOut += len(o.Conditions) + len(o.DependencyConditions) + len(o.Causes) + len(o.ReviewCauses)
			if in.Ready != o.Ready || in.DependencyReady != o.DependencyReady || in.ReviewPending != o.ReviewPending || in.LocalApproved != o.LocalApproved {
				r.BoolsMatch = false
			}
			for i, c := range o.Conditions {
				ic := in.Conditions[i]
				if c.Kind != ic.Kind || len(c.Path) != len(ic.Path) {
					t.Fatalf("%s: condition %d changed shape: %+v vs %+v", e.ID, i, c, ic)
				}
			}
		}
		rows = append(rows, r)
	}
	raw, _ := json.MarshalIndent(rows, "", "  ")
	if err := os.WriteFile(out, append(raw, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
